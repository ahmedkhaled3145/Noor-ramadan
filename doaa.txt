void saveDoaa () { 
ofstream file("doaa.txt", ios::out | ios::binary);
file << doaaCount << endl;
for (int i = 0; i < doaaCount; i++) {
file << doaalist[i].doaaID << endl;
file << doaalist[i].content << endl; 
file << doaalist[i].category << endl;
}
file.close(); 
}
